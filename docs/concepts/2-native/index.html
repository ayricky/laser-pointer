// A smaller first step. A better first week.
import { track } from './analytics';

type Milestone = {
  name: string;
  completed: boolean;
};

export async function activateWorkspace(
  workspaceId: string
) {
  const steps = await getMilestones(workspaceId);
  const completed = steps.filter(step => step.completed);

  if (completed.length >= 3) {
    await track('workspace_activated', {
      workspaceId,
      timeToValue: elapsedMinutes(),
      source: 'guided_setup'
    });
  }

  return {
    progress: completed.length / steps.length,
    next: steps.find(step => !step.completed)
  };
}
